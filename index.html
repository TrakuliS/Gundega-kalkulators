# uztura_kalkulators.py # Uztura kalkulators (CLI + drošs sandbox režīms) # Darbojas arī vidēs bez interaktīva input() # Balstīts uz Excel formulas 2.2 metodoloģiju """ LIETOŠANA (lokāli): python uztura_kalkulators.py Ja vide neatbalsta input() (piem., Replit, Colab, sandbox, CI), programma automātiski parāda DEMO rezultātus. """ import sys # ========================= # PALĪGFUNKCIJAS IEVADĒM # ========================= def ask_float(prompt, min_val=None, max_val=None): while True: try: value = float(input(prompt)) if min_val is not None and value < min_val: raise ValueError("Pārāk mazs skaitlis!") if max_val is not None and value > max_val: raise ValueError("Pārāk liels skaitlis!") return value except ValueError as e: print(f"❌ Nederīga ievade: {e}. Mēģini vēlreiz.") def ask_choice(prompt, choices): while True: print(prompt) for key, label in choices.items(): print(f" {key} – {label}") choice = input("Izvēle: ").strip() if choice in choices: return choice print("❌ Nederīga izvēle. Mēģini vēlreiz.") # ========================= # APRĒĶINU KODOLS (bez I/O) # ========================= def simulate(sex, age, height, weight, activity_factor, protein_g_per_kg, fat_percent): bmi = weight / ((height / 100) ** 2) # Ideālais svars (Devine formula) if sex == "Vīrietis": ideal_weight = 50 + 2.3 * ((height - 152.4) / 2.54) else: ideal_weight = 45.5 + 2.3 * ((height - 152.4) / 2.54) adjusted_weight = ideal_weight + 0.4 * (weight - ideal_weight) # BMR (Mifflin–St Jeor) – izmantojam adjusted_weight pie augsta BMI bmr_weight = adjusted_weight if bmi >= 35 else weight if sex == "Vīrietis": bmr = 10 * bmr_weight + 6.25 * height - 5 * age + 5 else: bmr = 10 * bmr_weight + 6.25 * height - 5 * age - 161 # TDEE tdee = bmr * activity_factor # Kaloriju mērķis (deficīts pie liekā svara) calories = max(tdee - 500, 1500) if bmi > 25 else tdee # Makroelementi protein_g = protein_g_per_kg * (adjusted_weight if bmi > 30 else weight) fat_kcal = calories * (fat_percent / 100) carb_kcal = calories - protein_g * 4 - fat_kcal return { "bmi": bmi, "bmr": bmr, "calories": calories, "protein_g": protein_g, "fat_g": fat_kcal / 9, "carb_g": carb_kcal / 4, } # ========================= # INTERAKTĪVĀ CLI DAĻA # ========================= def calculate_cli(): print("\n" + "="*30) print(" UZTURA KALKULATORS ") print("="*30 + "\n") sex_key = ask_choice("Dzimums:", {"1": "Sieviete", "2": "Vīrietis"}) sex = "Sieviete" if sex_key == "1" else "Vīrietis" age = ask_float("Vecums (gadi): ", 18, 100) height = ask_float("Augums (cm): ", 140, 220) weight = ask_float("Svars (kg): ", 40, 250) activity_key = ask_choice( "Fiziskā aktivitāte:", { "1": "Mazkustīgs (1.2)", "2": "Viegla aktivitāte (1.375)", "3": "Vidēja aktivitāte (1.55)", "4": "Augsta aktivitāte (1.725)", "5": "Ļoti augsta aktivitāte (1.9)", } ) activity_factor = {"1": 1.2, "2": 1.375, "3": 1.55, "4": 1.725, "5": 1.9}[activity_key] protein_g_per_kg = ask_float("Proteīni (g/kg): ", 1.0, 3.0) fat_percent = ask_float("Tauki (% no kcal): ", 15, 40) result = simulate( sex=sex, age=age, height=height, weight=weight, activity_factor=activity_factor, protein_g_per_kg=protein_g_per_kg, fat_percent=fat_percent, ) print("\n" + "-"*30) print(" REZULTĀTI ") print("-"*30) print(f"• BMI: {result['bmi']:.1f}") print(f"• BMR (pamat vielmaiņa): {result['bmr']:.0f} kcal") print(f"• Dienas kalorijas: {result['calories']:.0f} kcal") print("\nMakroelementi dienā:") print(f"• Proteīni: {result['protein_g']:.0f} g") print(f"• Tauki: {result['fat_g']:.0f} g") print(f"• Ogļhidrāti: {result['carb_g']:.0f} g") print("-"*30) # ========================= # DEMO REŽĪMS (online vidēm) # ========================= def show_demo(): print("\n" + "="*50) print(" DEMO REŽĪMS (input() nav pieejams šajā vidē) ") print("="*50 + "\n") demo = simulate( sex="Vīrietis", age=30, height=180, weight=85, activity_factor=1.55, protein_g_per_kg=1.6, fat_percent=30, ) print("Piemērs: Vīrietis, 30 gadi, 180 cm, 85 kg, vidēja aktivitāte") print("-"*60) print(f"• BMI: {demo['bmi']:.1f}") print(f"• BMR: {demo['bmr']:.0f} kcal") print(f"• Dienas kalorijas: {demo['calories']:.0f} kcal") print(f"• Proteīni: {demo['protein_g']:.0f} g") print(f"• Tauki: {demo['fat_g']:.0f} g") print(f"• Ogļhidrāti: {demo['carb_g']:.0f} g") print("-"*60) # ========================= # ENTRY POINT # ========================= FORCE_DEMO_MODE = False # Maini uz False, ja gribi interaktīvu režīmu tur, kur iespējams if __name__ == "__main__": # Pārbaudām, vai ir interaktīvs terminālis is_interactive = sys.stdin is not None and sys.stdin.isatty() if FORCE_DEMO_MODE or not is_interactive: show_demo() else: try: calculate_cli() except KeyboardInterrupt: print("\n\nProgramma pārtraukta. Uz redzēšanos!")
